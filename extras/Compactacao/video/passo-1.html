public Usuario atualizarGerenciado(String emailAtor, Long id, UsuarioForm form) { Usuario alvo = buscarGerenciado(emailAtor, id); if (usuarios.existsByEmailIgnoreCaseAndIdNot(form.getEmail(), id)) throw new IllegalArgumentException("Já existe uma conta com este e-mail."); if (form.getTipoUsuario() != alvo.getTipoUsuario()) throw new IllegalArgumentException("O tipo do usuário não pode ser alterado."); if (StringUtils.hasText(form.getSenha()) && form.getSenha().length() < 8) throw new IllegalArgumentException("A senha precisa ter pelo menos 8 caracteres."); preencherBase(alvo, form); preencherPerfil(alvo, form); if (StringUtils.hasText(form.getSenha())) alvo.setSenha(passwordEncoder.encode(form.getSenha())); return usuarios.save(alvo); }
